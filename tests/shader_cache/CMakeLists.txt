cmake_minimum_required(VERSION 3.24)
project(ShaderCacheTests LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "These tests exercise Windows shader cache replacement")
endif()

# Compile the production IO routines, without linking the renderer or requiring game assets.
# Like tools/shader-validate, extract the implementation rather than maintain a test copy.
set(BACKEND "${CMAKE_CURRENT_SOURCE_DIR}/../../src/fast/backends/gfx_direct3d11.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BACKEND}")
file(READ "${BACKEND}" SOURCE)
string(FIND "${SOURCE}" "constexpr uint32_t kShaderCacheMagic" HEADER_START)
string(FIND "${SOURCE}" "// Empty when the cache directory" HEADER_END)
string(FIND "${SOURCE}" "bool ShaderCacheLoad(" IO_START)
if(HEADER_START LESS 0 OR HEADER_END LESS HEADER_START OR IO_START LESS HEADER_END)
    message(FATAL_ERROR "Shader cache extraction markers changed")
endif()
math(EXPR HEADER_LENGTH "${HEADER_END} - ${HEADER_START}")
string(SUBSTRING "${SOURCE}" ${HEADER_START} ${HEADER_LENGTH} HEADER)
string(SUBSTRING "${SOURCE}" ${IO_START} -1 IO_REST)
string(FIND "${IO_REST}" "} // namespace" IO_LENGTH)
if(IO_LENGTH LESS 0)
    message(FATAL_ERROR "Shader cache IO end marker missing")
endif()
string(SUBSTRING "${IO_REST}" 0 ${IO_LENGTH} IO)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shader_cache_io.inc" "${HEADER}\n${IO}")

add_executable(shader_cache_tests shader_cache_tests.cpp)
target_compile_features(shader_cache_tests PRIVATE cxx_std_20)
target_compile_definitions(shader_cache_tests PRIVATE NOMINMAX)
target_include_directories(shader_cache_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME shader_cache COMMAND shader_cache_tests "${CMAKE_CURRENT_BINARY_DIR}/cache-fixture")
set_tests_properties(shader_cache PROPERTIES TIMEOUT 30)
