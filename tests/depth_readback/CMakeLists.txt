cmake_minimum_required(VERSION 3.24)
project(DepthReadbackTests LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This test uses the Windows Direct3D 11 WARP device")
endif()
add_executable(depth_readback_tests depth_readback_tests.cpp)
target_compile_features(depth_readback_tests PRIVATE cxx_std_20)
target_include_directories(depth_readback_tests PRIVATE ../../include)
target_compile_definitions(depth_readback_tests PRIVATE NOMINMAX)
target_link_libraries(depth_readback_tests PRIVATE d3d11)
enable_testing()
add_test(NAME depth_readback COMMAND depth_readback_tests)
set_tests_properties(depth_readback PROPERTIES TIMEOUT 30)
